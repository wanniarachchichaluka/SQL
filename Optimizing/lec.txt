#Transaction = unit of work in a database
#Properties of transaction - ACID
#A-Atomicity
#C-Consistency
#I-Isolation
#D-Durability

#Scenario:-Alice giving 10 to Bob
##need to to use two quries without somebody seeing in mediate of transaction
#Atomicity
sqlite> BEGIN TRANSACTION;
sqlite> UPDATE "accounts" SET "balance" = "balance" + 10
   ...> WHERE "id" = 2;
sqlite> UPDATE "accounts" SET "balance" = "balance" - 10
   ...> WHERE "id" = 1;  
sqlite> COMMIT;
sqlite> SELECT * FROM "accounts";
1|Alice|0
2|Bob|30
3|Charlie|30
sqlite> 

#Alice paying 10 more to Bob
sqlite> SELECT * FROM "accounts";
1|Alice|0
2|Bob|30
3|Charlie|30
sqlite> BEGIN TRANSACTION;
sqlite> UPDATE "accounts" SET "balance" = "balance" + 10
   ...> WHERE "id" = 2;
sqlite> UPDATE "accounts" SET "balance" = "balance" - 10
   ...> WHERE "id" = 1;
Runtime error: CHECK constraint failed: balance (19)
sqlite> ROLLBACK;                                       
sqlite> SELECT * FROM "accounts";
1|Alice|0
2|Bob|30
3|Charlie|30
sqlite> 
##No harm done because of rollback in above
